# Corfu guide - static single-page site. Everything here is public
# content meant to be indexed; there is no admin area, no user-generated
# content, and no private routes to exclude.
User-agent: *
Allow: /

# Local-only dev/build files that are not part of the published site.
# Harmless if they are never deployed, but explicitly excluded so they
# can't turn up in an index if they are.
Disallow: /scripts/
Disallow: /schema.sql
Disallow: /package.json
Disallow: /tailwind.config.js
